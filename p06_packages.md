# Install libraries in generated projects

Work in `projectgen-src/`, with its environment active.
Keep the src layout and extend the existing generator.

## 1. Behaviour

```bash
projectgen text-analysis --data --notebooks -p requests
```

- create the selected project structure and `env/`
- install selected packages using the Python interpreter inside the new `env/`
- simple: write selected requirements to `requirements.txt`
- flat/src: write selected requirements to `[project].dependencies`
- no `-p`: install no extra libraries
- `-n` with `-p`: reject before creating the project

These are dependencies of the generated project. The generator itself still uses only the standard library.
The recorded list contains the requested requirements, not a lock file of all installed versions.

## 2. Update four modules

Keep `naming.py` and `structure.py` unchanged.

### `src/projectgen/environment.py`

```python
import os
import subprocess
import venv


def create_environment(project):
    venv.create(project / 'env', with_pip=True)


def install_packages(project, packages):
    if os.name == 'nt':
        python = project / 'env' / 'Scripts' / 'python.exe'
    else:
        python = project / 'env' / 'bin' / 'python'

    subprocess.run(
        [str(python), '-m', 'pip', 'install', *packages],
        check=True
    )
```

### `src/projectgen/myparser.py`

```python
import argparse

from .naming import project_name


def parse_arguments():
    parser = argparse.ArgumentParser(description='Create a Python project')
    parser.add_argument(
        'name', nargs='?', type=project_name,
        help='Project name (default: project-YYYYMMDDHHMMSS)'
    )
    parser.add_argument(
        '-t', '--type', choices=['simple', 'flat', 'src'], default='src',
        help='Project type (default: src)'
    )
    parser.add_argument(
        '-d', '--directory', default='.',
        help='Existing parent directory (default: current directory)'
    )
    parser.add_argument(
        '-n', '--no-venv', action='store_true',
        help='Do not create a virtual environment'
    )
    parser.add_argument('--data', action='store_true', help='Create data directory')
    parser.add_argument(
        '--notebooks', action='store_true', help='Create notebooks directory'
    )
    parser.add_argument(
        '-p', '--packages', nargs='+', default=[],
        help='Packages to install in the new project environment'
    )
    args = parser.parse_args()
    if args.no_venv and args.packages:
        parser.error('--packages requires a virtual environment; remove -n')
    return args
```

### `src/projectgen/projectfiles.py`

```python
import json


def create_package_cli(package_dir):
    content = """def main():
    pass


if __name__ == '__main__':
    main()
"""
    (package_dir / 'cli.py').write_text(content, encoding='utf-8')


def create_readme(project, name):
    (project / 'README.md').write_text(f'# {name}\n', encoding='utf-8')


def create_pyproject(project, name, package, where, packages):
    content = f'''[build-system]
requires = ["setuptools>=77.0.3"]
build-backend = "setuptools.build_meta"

[project]
name = "{name}"
version = "0.1.0"
description = "Python project generated by projectgen"
requires-python = ">=3.10"
dependencies = {json.dumps(packages)}

[project.scripts]
"{name}" = "{package}.cli:main"

[tool.setuptools.packages.find]
where = ["{where}"]
include = ["{package}", "{package}.*"]
'''
    (project / 'pyproject.toml').write_text(content, encoding='utf-8')


def create_requirements(project, packages):
    content = '\n'.join(packages) + '\n'
    (project / 'requirements.txt').write_text(content, encoding='utf-8')
```

### `src/projectgen/cli.py`

```python
from pathlib import Path

from .environment import create_environment, install_packages
from .myparser import parse_arguments
from .naming import default_name, package_name
from .projectfiles import create_package_cli, create_pyproject, create_readme
from .projectfiles import create_requirements
from .structure import create_flat, create_optional_directories, create_simple
from .structure import create_src


def main():
    args = parse_arguments()
    name = args.name or default_name()
    project = Path(args.directory).expanduser().resolve() / name

    if args.type == 'simple':
        create_simple(project)
        if args.packages:
            create_requirements(project, args.packages)

    else:
        package = package_name(name)
        if args.type == 'flat':
            package_dir = create_flat(project, package)
            where = '.'
        else:
            package_dir = create_src(project, package)
            where = 'src'
        create_package_cli(package_dir)
        create_readme(project, name)
        create_pyproject(project, name, package, where, args.packages)

    create_optional_directories(project, args.data, args.notebooks)
    if not args.no_venv:
        create_environment(project)
        if args.packages:
            install_packages(project, args.packages)

    print(f'Created {args.type} project: {project}')


if __name__ == '__main__':
    main()
```

`subprocess.run(..., check=True)` stops with an error if installation fails.
The partly created project remains available for inspection.
Activating the new environment is unnecessary: its interpreter is called by its path.

## 3. Test

The generator is installed in editable mode; these Python code changes need no reinstall.

```bash
projectgen -h
projectgen test_packages_src -p requests
projectgen test_packages_simple -t simple -p requests
projectgen test_packages_invalid -n -p requests
```

Check:

- src TOML contains `dependencies = ["requests"]`
- simple `requirements.txt` contains `requests`
- invalid combination prints an argument error; `test_packages_invalid/` does not exist

Verify installation in the generated environment.

Windows PowerShell:

```powershell
.\test_packages_src\env\Scripts\python.exe -m pip show requests
```

Linux / macOS:

```bash
./test_packages_src/env/bin/python -m pip show requests
```

For version constraints, quote each requirement:

```bash
projectgen test_packages_version -p "requests>=2.31,<3"
```

Put the project name before `-p`; subsequent values belong to the package list.
Installing a generated flat/src project later also installs its declared dependencies.
For simple, use `python -m pip install -r requirements.txt` in its environment.
